Create your first sceneTạo scene đầu tiên→
first-scene.ts
import * as THREE from 'three';
import {
  createPipeline, onFrame,
  setRenderFunction, startLoop,
} from '@tunn/oxygen';

startLoop();
const host = document.getElementById('scene');
if (!host) throw new Error('Scene host not found');
const pipeline = createPipeline(host);
const camera = new THREE.PerspectiveCamera(45, 1, 0.1, 100);
camera.position.z = 4;

const mesh = new THREE.Mesh(
  new THREE.TorusKnotGeometry(),
  new THREE.MeshNormalMaterial(),
);
pipeline.scene.add(mesh);
onFrame((_time, delta) => { mesh.rotation.y += delta * 0.001; });
setRenderFunction(() => {
  camera.aspect = pipeline.size.width / pipeline.size.height;
  camera.updateProjectionMatrix();
  pipeline.render(camera);
});
